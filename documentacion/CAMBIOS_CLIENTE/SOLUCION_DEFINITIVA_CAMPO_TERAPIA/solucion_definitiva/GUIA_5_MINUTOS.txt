from django.db import models
from apps.terapias.models import Terapia  # ← Agregar import

class GrupoTerapeutico(models.Model):

    # Campos existentes
    nombre = models.CharField(max_length=200, verbose_name='Nombre del Grupo')
    descripcion = models.TextField(blank=True, verbose_name='Descripción')
    
    # ✅ NUEVO CAMPO (Agregar aquí):
    terapia = models.ForeignKey(
        Terapia,
        on_delete=models.PROTECT,
        related_name='grupos_terapeuticos',
        verbose_name='Terapia',
        null=True,
        blank=True,
        help_text='Tipo de terapia que ofrece este grupo'
    )
    
    # Otros campos existentes
    capacidad_maxima = models.IntegerField(default=10)
    cupos_disponibles = models.IntegerField(default=10)
    activo = models.BooleanField(default=True)
    fecha_creacion = models.DateTimeField(auto_now_add=True)
    # ... más campos ...
    
    class Meta:
        verbose_name = 'Grupo Terapéutico'
        verbose_name_plural = 'Grupos Terapéuticos'
    
    def __str__(self):
        return self.nombre

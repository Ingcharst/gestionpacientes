**REEMPLAZAR con:**
```python
def asignar_desde_recomendacion(request, paciente_id, grupo_id):
    """
    Asignar paciente a grupo desde recomendaciones
    """
    from apps.procedimientos.models import Paciente, AdmisionTerapia
    from datetime import date
    
    paciente = get_object_or_404(Paciente, pk=paciente_id)
    grupo = get_object_or_404(GrupoTerapeutico, pk=grupo_id)
    
    try:
        # Obtener admisión vigente del paciente para esta terapia
        fecha_hoy = date.today()
        admision = AdmisionTerapia.objects.filter(
            paciente=paciente,
            terapia=grupo.terapia,
            estado='VIGENTE',
            fecha_inicio__lte=fecha_hoy,
            fecha_fin__gte=fecha_hoy
        ).first()
        
        if not admision:
            messages.error(
                request,
                f'El paciente no tiene una admisión vigente para {grupo.terapia.nombre}'
            )
            return redirect('grupos:recomendar_grupos', paciente_id=paciente_id)
        
        # Crear asignación
        asignacion = AsignacionGrupo.objects.create(
            paciente=paciente,
            grupo=grupo,
            admision_terapia=admision,
            dias_asistencia=grupo.dias_disponibles,  # O los días que necesites
            numero_terapias_semanales=3,  # Ajustar según necesidad
            estado='ACTIVA'
        )
        
        messages.success(
            request,
            f'✅ {paciente.nombre_completo} asignado exitosamente a {grupo.nombre}'
        )
        
        # ✅ REDIRIGIR al detalle del grupo
        return redirect('grupos:detalle_grupo', pk=grupo_id)
        
    except Exception as e:
        messages.error(request, f'Error al asignar: {str(e)}')
        return redirect('grupos:recomendar_grupos', paciente_id=paciente_id)
```
